cmake_minimum_required(VERSION 3.16)
project(dag_chain_solver_v8 VERSION 0.8.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(chain_algorithm STATIC src/chain.cpp src/refine.cpp src/regions.cpp src/slack.cpp src/anytime.cpp src/spill.cpp src/tensor_neighborhood.cpp)
target_include_directories(chain_algorithm PUBLIC src)
if(UNIX)
    # Use standard C++ threads without find_package or external dependencies.
    target_compile_options(chain_algorithm PUBLIC -pthread)
    target_link_options(chain_algorithm PUBLIC -pthread)
endif()
if(MSVC)
    target_compile_options(chain_algorithm PUBLIC /utf-8 /W4)
else()
    target_compile_options(chain_algorithm PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(npu_sched src/main.cpp)
target_link_libraries(npu_sched PRIVATE chain_algorithm)
if(MINGW)
    target_link_options(npu_sched PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
