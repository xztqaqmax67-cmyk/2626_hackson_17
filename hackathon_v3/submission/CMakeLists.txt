cmake_minimum_required(VERSION 3.16)
project(npu_sched LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(npu_sched src/solver.cpp)
target_include_directories(npu_sched PRIVATE src )
set_target_properties(npu_sched PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" config_upper)
  set_target_properties(npu_sched PROPERTIES "RUNTIME_OUTPUT_DIRECTORY_${config_upper}" "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(MSVC)
  target_compile_options(npu_sched PRIVATE /utf-8 /W4 /permissive-)
else()
  target_compile_options(npu_sched PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(npu_sched PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()
