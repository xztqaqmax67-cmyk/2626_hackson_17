cmake_minimum_required(VERSION 3.16)
project(dag_chain_solver_v3 VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(chain_algorithm STATIC src/chain.cpp)
target_include_directories(chain_algorithm PUBLIC src)
if(UNIX)
    # Use standard C++ threads without find_package or external dependencies.
    target_compile_options(chain_algorithm PUBLIC -pthread)
    target_link_options(chain_algorithm PUBLIC -pthread)
endif()
if(MSVC)
    target_compile_options(chain_algorithm PUBLIC /utf-8 /W4)
else()
    target_compile_options(chain_algorithm PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(npu_sched src/main.cpp)
target_link_libraries(npu_sched PRIVATE chain_algorithm)
if(MINGW)
    target_link_options(npu_sched PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
option(CHAIN_BUILD_TESTS "Build development-only tests and comparison baseline" OFF)
if(CHAIN_BUILD_TESTS)
    enable_testing()
    add_executable(chain_tests tests/chain_tests.cpp)
    target_link_libraries(chain_tests PRIVATE chain_algorithm)
    add_test(NAME chain_invariants COMMAND chain_tests)
    add_executable(parallel_jobs_tests tests/parallel_jobs_tests.cpp)
    target_link_libraries(parallel_jobs_tests PRIVATE chain_algorithm)
    add_test(NAME parallel_jobs COMMAND parallel_jobs_tests)
    add_executable(rules_memory_tests tests/rules_memory_tests.cpp)
    target_link_libraries(rules_memory_tests PRIVATE chain_algorithm)
    add_test(NAME rules_memory COMMAND rules_memory_tests)
    add_executable(rules_contract_tests tests/rules_contract_tests.cpp)
    target_link_libraries(rules_contract_tests PRIVATE chain_algorithm)
    add_test(NAME rules_contract COMMAND rules_contract_tests)
    add_executable(chain_baseline src/main.cpp)
    target_compile_definitions(chain_baseline PRIVATE CHAIN_BASELINE=1)
    target_link_libraries(chain_baseline PRIVATE chain_algorithm)
    if(MINGW)
        target_link_options(chain_tests PRIVATE -static -static-libgcc -static-libstdc++)
        target_link_options(parallel_jobs_tests PRIVATE -static -static-libgcc -static-libstdc++)
        target_link_options(rules_memory_tests PRIVATE -static -static-libgcc -static-libstdc++)
        target_link_options(rules_contract_tests PRIVATE -static -static-libgcc -static-libstdc++)
        target_link_options(chain_baseline PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
endif()
